```sql

SELECT 
	InvoiceDate AS [Invoice Date],
	BillingAddress AS [Billing Address],
	BillingCity AS [Billing City],
	Total AS Amount,
	CASE 
		WHEN Total < 2.00 THEN 'Baseline'
		WHEN Total BETWEEN 2.00 AND 6.99 THEN 'Low'
		WHEN Total BETWEEN 7.00 AND 15.00 THEN 'Target'
		ELSE 'Top'
	END
		AS PurchaseType
FROM 
	Invoice as i
WHERE
		PurchaseType = 'Top'
	OR
		PurchaseType = 'Target'
ORDER BY
	InvoiceDate ASC;
```
