```sql
WITH invoice_dates AS (
    SELECT
        InvoiceDate
    FROM
        Invoice
    WHERE
        InvoiceId IN (251, 252, 254)
)
SELECT
    i.InvoiceId,
    i.InvoiceDate,
    i.BillingAddress,
    i.BillingCity
FROM 
    Invoice AS i 
WHERE
    i.InvoiceDate IN invoice_dates
ORDER BY
    i.InvoiceDate ASC;
```
