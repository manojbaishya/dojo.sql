```sql

SELECT 
	InvoiceDate,
	BillingAddress,
	BillingCity,
	Total
FROM 
	Invoice as i 
WHERE 
	BillingCity = 'Brussels'
ORDER BY
	InvoiceDate;
```

```sql
SELECT 
	InvoiceDate,
	BillingAddress,
	BillingCity,
	Total

	Invoice as i
```
