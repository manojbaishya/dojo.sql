```sql
-- Invoices in cities starting with B

SELECT 
	InvoiceDate,
	BillingAddress,
	BillingCity,
	Total
FROM 
	Invoice as i 
WHERE 
	BillingCity LIKE 'B%'
ORDER BY
	InvoiceDate;
```


```
SELECT 
	InvoiceDate,
	BillingAddress,
	BillingCity,
	Total
FROM 
	Invoice as i 
WHERE 
	BillingCity LIKE '%B%' or BillingCity LIKE '%b%'
ORDER BY
	InvoiceDate;
```
