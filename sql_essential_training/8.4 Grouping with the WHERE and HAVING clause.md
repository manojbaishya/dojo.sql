```sql
SELECT 
    i.BillingCity AS [Billing City],
    ROUND(AVG(i.Total), 2) AS [Average Purchase] 
FROM 
    Invoice AS i
WHERE 
    [Billing City] IS NOT NULL
    AND (
        [Billing City] LIKE 'B%' 
        OR [Billing City] LIKE 'b%'
        )
GROUP BY 
    [Billing City]
HAVING 
    [Average Purchase] > 5.00
ORDER BY 
    [Average Purchase] DESC;
```
