```sql
WITH date_criteria AS (
    SELECT
        InvoiceDate
    FROM
        Invoice
    WHERE
        InvoiceId = 251
)
SELECT
    i.InvoiceDate,
    i.BillingAddress,
    i.BillingCity
FROM Invoice AS i
    INNER JOIN date_criteria
WHERE
    i.InvoiceDate > date_criteria.InvoiceDate
ORDER BY
    i.InvoiceDate ASC;
```
