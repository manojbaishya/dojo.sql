	InvoiceDate,
	BillingAddress,
	BillingCity,
	Total
FROM 
	Invoice as i

	DATE(InvoiceDate) = '2010-05-22'
	--InvoiceDate = '2010-05-22 00:00:00'
	--InvoiceDate BETWEEN '2010-05-22 00:00:00' AND '2010-05-30 00:00:00'

	InvoiceDate;
