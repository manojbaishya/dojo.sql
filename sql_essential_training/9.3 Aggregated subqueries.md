-- How is each individual city performing against the global average sales?

-- Solution 1
```sql
WITH Averages AS (
    SELECT
        BillingCity,
        ROUND(AVG(Total), 2) AS CityAverage,
        ROUND((
                SELECT 
                    AVG(Total) FROM Invoice
                WHERE
                    BillingCity IS NOT NULL
            ), 2) 
        AS GlobalAverage
    FROM
        Invoice
    WHERE
        BillingCity IS NOT NULL
    GROUP BY
        BillingCity
    )
SELECT 
    i.BillingCity AS [Billing City],
    Averages.CityAverage AS [City Average],
    Averages.GlobalAverage AS [Global Average],
    ROUND((Averages.CityAverage - Averages.GlobalAverage) / Averages.GlobalAverage * 100, 2) AS [Sales Lead Percent]
FROM
    Invoice AS i
    INNER JOIN Averages
        ON Averages.BillingCity = i.BillingCity
WHERE
    i.BillingCity IS NOT NULL
GROUP BY
    i.BillingCity
ORDER BY
    [City Average] DESC;
```

-- Alternate query / Solution 2
-- INNER JOIN on 1=1 or true

```sql
WITH c AS (
    SELECT
        BillingCity,
        AVG(Total) AS c_avg
    FROM
        Invoice
    WHERE
        BillingCity IS NOT NULL
    GROUP BY
        BillingCity
),
g AS (
    SELECT 
        AVG(Total) AS g_avg
    FROM 
        Invoice
    WHERE
        BillingCity IS NOT NULL
)
SELECT 
    c.BillingCity AS [Billing City],
    ROUND(c.c_avg, 2) AS [City Average in USD],
    ROUND(g.g_avg, 2) AS [Global Average  in USD],
    ROUND((c.c_avg - g.g_avg) / g.g_avg * 100, 2) AS [Sales Lead Percent]
FROM
    c
    INNER JOIN g
        ON true
ORDER BY
    [City Average in USD] DESC;
```
