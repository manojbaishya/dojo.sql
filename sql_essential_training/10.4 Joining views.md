```sql
CREATE VIEW V_Tracks_InvoiceLine AS
SELECT 
    t.Name,
    t.Composer,
    t.Milliseconds / 1000 AS [Song Length (Seconds)],
    il.UnitPrice,
    il.Quantity,
    il.Quantity * il.UnitPrice AS [Amount]
FROM 
    InvoiceLine AS il 
    INNER JOIN Track AS t 
        ON t.TrackId = il.TrackId
ORDER BY
    [Amount] DESC;
```
